01Empty Graph

A simple undirected graph G = (V, E) has n = |V| vertices and m = |E| edges. Each edge is an unordered pair of distinct vertices. The empty graph has no edges, so each vertex is isolated and forms its own connected component. Write c for the number of components.

Reproduce the figures in Python

The March 30 generator uses Python, NetworkX, Matplotlib and Pillow. Run it from the directory where you want its output folder. It writes 20 PNGs, four GIFs and a manifest of the files. Optional MP4 export uses FFmpeg.

#!/usr/bin/env python3

import json
import os
from dataclasses import dataclass
from typing import Callable, Dict, List, Optional, Tuple

import matplotlib.pyplot as plt
import networkx as nx
from matplotlib.animation import FuncAnimation


OUTPUT_DIR = "2026-03-30-basic-graph-animations"
SAVE_MP4 = False
SAVE_GIF = True
DPI = 220
FIGSIZE = (6.4, 7.2)

BG = "#fbfaf7"
EDGE = "#777777"
NODE = "#4c78a8"
NODE_ALT = "#72b7b2"
NODE_HI = "#e45756"
NODE_ACCENT = "#f2cf5b"
TEXT = "#1f1f1f"
SUBTEXT = "#585858"


@dataclass
class VisualSpec:
    slug: str
    title: str
    description: str
    kind: str
    builder: Callable[[], Dict]


def ensure_dir(path: str):
    os.makedirs(path, exist_ok=True)


def circular_pos(nodes):
    G = nx.Graph()
    G.add_nodes_from(nodes)
    return nx.circular_layout(G)


def bipartite_pos(left: List[int], right: List[int]) -> Dict[int, Tuple[float, float]]:
    pos = {}
    left_y = list(reversed(range(len(left))))
    right_y = list(reversed(range(len(right))))
    for i, u in enumerate(left):
        pos[u] = (-1.0, left_y[i])
    for i, v in enumerate(right):
        pos[v] = (1.0, right_y[i])
    return pos


def shell_pos(shells: List[List[int]]) -> Dict[int, Tuple[float, float]]:
    G = nx.Graph()
    for shell in shells:
        G.add_nodes_from(shell)
    return nx.shell_layout(G, shells)


def draw_base(
    ax_graph,
    G: nx.Graph,
    pos: Dict,
    title: str,
    description: str,
    node_colors: Optional[List[str]] = None,
    edge_colors: Optional[List[str]] = None,
    edge_widths: Optional[List[float]] = None,
    node_size: int = 520,
    font_size: int = 10,
):
    ax_graph.clear()
    ax_graph.set_facecolor(BG)

    if node_colors is None:
        node_colors = [NODE for _ in G.nodes()]
    if edge_colors is None:
        edge_colors = [EDGE for _ in G.edges()]
    if edge_widths is None:
        edge_widths = [1.8 for _ in G.edges()]

    nx.draw_networkx_nodes(
        G, pos, ax=ax_graph,
        node_color=node_colors,
        node_size=node_size,
        linewidths=0.7,
        edgecolors="#ffffff"
    )
    nx.draw_networkx_edges(
        G, pos, ax=ax_graph,
        edge_color=edge_colors,
        width=edge_widths
    )
    nx.draw_networkx_labels(
        G, pos, ax=ax_graph,
        labels={u: str(u) for u in G.nodes()},
        font_size=font_size,
        font_color="#ffffff"
    )

    ax_graph.set_title(title, fontsize=16, color=TEXT, pad=14)
    ax_graph.set_axis_off()

    ax_graph.text(
        0.5, -0.08, description,
        transform=ax_graph.transAxes,
        ha="center", va="top",
        fontsize=10.2,
        color=SUBTEXT,
        wrap=True
    )


def save_static(spec: VisualSpec, payload: Dict):
    fig, ax = plt.subplots(figsize=FIGSIZE)
    fig.patch.set_facecolor(BG)

    draw_base(
        ax_graph=ax,
        G=payload["G"],
        pos=payload["pos"],
        title=spec.title,
        description=spec.description,
        node_colors=payload.get("node_colors"),
        edge_colors=payload.get("edge_colors"),
        edge_widths=payload.get("edge_widths"),
        node_size=payload.get("node_size", 520),
        font_size=payload.get("font_size", 10),
    )

    out_png = os.path.join(OUTPUT_DIR, f"{spec.slug}.png")
    plt.tight_layout()
    plt.savefig(out_png, dpi=DPI, bbox_inches="tight", facecolor=BG)
    plt.close(fig)
    return {"png": out_png}


def save_dynamic(spec: VisualSpec, payload: Dict):
    fig, ax = plt.subplots(figsize=FIGSIZE)
    fig.patch.set_facecolor(BG)

    G_base: nx.Graph = payload["G"]
    pos = payload["pos"]
    frames = payload["frames"]
    node_size = payload.get("node_size", 520)
    font_size = payload.get("font_size", 10)

    def update(frame_index: int):
        frame = frames[frame_index]
        G_now = nx.Graph()
        G_now.add_nodes_from(G_base.nodes())
        G_now.add_edges_from(frame["edges"])

        node_colors = frame.get("node_colors")
        if node_colors is None:
            node_colors = [NODE for _ in G_now.nodes()]

        edge_colors = frame.get("edge_colors")
        if edge_colors is None:
            edge_colors = [EDGE for _ in G_now.edges()]

        edge_widths = frame.get("edge_widths")
        if edge_widths is None:
            edge_widths = [1.8 for _ in G_now.edges()]

        desc = spec.description
        if "caption" in frame and frame["caption"]:
            desc = frame["caption"]

        draw_base(
            ax_graph=ax,
            G=G_now,
            pos=pos,
            title=spec.title,
            description=desc,
            node_colors=node_colors,
            edge_colors=edge_colors,
            edge_widths=edge_widths,
            node_size=node_size,
            font_size=font_size,
        )

    anim = FuncAnimation(
        fig,
        update,
        frames=len(frames),
        interval=900,
        repeat=True
    )

    outputs = {}

    if SAVE_GIF:
        out_gif = os.path.join(OUTPUT_DIR, f"{spec.slug}.gif")
        anim.save(out_gif, writer="pillow", fps=1)
        outputs["gif"] = out_gif

    if SAVE_MP4:
        out_mp4 = os.path.join(OUTPUT_DIR, f"{spec.slug}.mp4")
        try:
            anim.save(out_mp4, writer="ffmpeg", fps=1)
            outputs["mp4"] = out_mp4
        except Exception:
            pass

    plt.close(fig)
    return outputs


def manifest_entry(spec: VisualSpec, outputs: Dict):
    return {
        "slug": spec.slug,
        "title": spec.title,
        "description": spec.description,
        "kind": spec.kind,
        "files": outputs,
    }

def d_triangle_free_vs_triangle():
    G = nx.empty_graph(8)
    pos = circular_pos(G.nodes())
    seq = [
        [(0,1),(1,2)],
        [(0,1),(1,2),(2,3),(3,4)],
        [(0,1),(1,2),(2,3),(3,4),(4,5)],
        [(0,1),(1,2),(2,0),(3,4),(4,5)]
    ]
    frames = []
    for i, edges in enumerate(seq):
        caption = "Triangle-free so far." if i < 3 else "Now a triangle appears on vertices 0, 1, 2."
        edge_colors = []
        for e in edges:
            if set(e).issubset({0,1,2}) and len(edges) == 5:
                edge_colors.append(NODE_HI)
            else:
                edge_colors.append(EDGE)
        frames.append({
            "edges": edges,
            "edge_colors": edge_colors,
            "caption": caption
        })
    return {"G": G, "pos": pos, "frames": frames}

02Single Edge

Join vertices 0 and 1. They are now adjacent, and their shared edge is incident to each of them. The degree deg(v) counts the edges incident to v. Its open neighbourhood N(v) is the set of its neighbours. Each edge contributes to the degrees at both ends, so the total degree is 2m.

Degrees and neighbours

Select a vertex in this larger example. Its highlighted edges connect it to its neighbours. Change the view to follow a shortest path, reverse the adjacency rule in a complement, or inspect an induced subgraph.

Inspect a vertex

From degree to the adjacency matrix

The adjacency matrix A(G) records which vertices meet. Entry Aᵢⱼ is 1 when an edge joins i to j and 0 otherwise. For the six-vertex explorer graph, the matrix is symmetric and each row sum is a degree.

Every edge contributes at both endpoints, giving the handshaking lemma

∑v ∈ V deg(v) = 2m.

03Path Graph

A path follows successive edges through distinct vertices. The path graph Pₙ is one chain of n vertices joined by n − 1 edges. A walk may revisit a vertex. For example, 0–1–0–1 is a walk of length three, while the path 0–1 has length one.

Lengths and weighted distances

Path length counts edges. When edges carry nonnegative costs, a least-cost path minimizes their sum and can use more edges than a path of minimum length.

How is the least-cost route found?

The experiment enumerates every simple A–D path and compares their total costs. With nonnegative weights, removing a cycle from a walk can only lower or preserve its cost. On larger graphs, Dijkstra’s algorithm avoids enumerating paths by repeatedly settling the unsettled vertex with the smallest tentative distance and updating its neighbours.

04Cycle Graph

Add 7–0 to P₈ to form a cycle. A simple cycle returns to its start with no other repeated vertex. Along Cₙ every vertex has degree two, so the degree sum 2n = 2m gives m = n.

Odd cycles, even cycles and colour

A proper colouring gives adjacent vertices different colours. The chromatic number χ(G) is the fewest colours that suffice. Alternate two colours along a cycle to see where parity matters.

05Complete Graph

A complete graph Kₙ contains one edge for every pair of distinct vertices. Once all pairs are joined, no more edges can be added while the graph stays simple.

Count pairs by their endpoints

m = n(n − 1)/2

There are n choices for the first endpoint and n − 1 for the second. Each unordered pair appears in both orders, so divide by two. In K₈ this gives 8 × 7 / 2 = 28.

How a clique grows

A clique is a set of pairwise adjacent vertices. The numbered vertex joins every earlier vertex. Each step adds one more edge than the preceding step.

06Star Graph

In a star K₁,ᵣ, one hub meets all r leaves. A leaf has degree one. Like P₈, the eight-vertex star has seven edges, yet concentrating those edges at a hub changes every long-distance route.

Removing the hub or a leaf

In every star with at least two leaves, each pair of leaves is two steps apart. The number of components after deleting the hub reveals how many leaves depend on it.

07Tree

A tree is a connected graph with no cycle. Connectivity supplies a path between each pair of vertices. Two different simple paths between the same endpoints would contain a cycle, so the path is unique.

Why does a tree have n − 1 edges?

m = n − 1

An endpoint of a longest simple path is a leaf. A second neighbour would extend the path or create a cycle. Removing the leaf and its edge leaves a tree. Repeat until one vertex and zero edges remain.

Path, star and rooted binary tree

Each example has seven vertices and six edges. In a rooted binary tree each vertex has at most two children. The branching example has all leaves at the same depth; the path rooted at an endpoint grows one level at a time.

Add a chord to the original tree

The existing path between two nonadjacent vertices is unique. Joining its endpoints closes exactly one cycle. Deleting any edge of the intact tree instead separates its endpoints.

08Connected Graph

A graph is connected when a path joins each pair of vertices. Extra edges can shorten paths while preserving connectivity. Here the chord 0–4 offers a one-edge route between vertices that are four steps apart along the rim.

Building Connectivity

Start with all eight vertices isolated. Join one more of them to the path at each step. The number of components falls by one at each step until the path spans all eight vertices.

09Disconnected Graph

A disconnected graph has at least two components. Each component is a maximal set of vertices joined by paths. No path can cross between components. A forest is an acyclic graph. Each component is a tree, including a single isolated vertex.

Count components in a forest

m = n − c

For component i with nᵢ vertices, its tree has nᵢ − 1 edges. Summing over all c components gives m = n − c. Any additional edge within a component contributes to the excess of m over n − c.

When do the components join?

A sampled graph can contain isolated vertices, trees and components with cycles. Hold the seed fixed and increase the probability to see when components merge. A new edge inside one component leaves c unchanged.

10Complete Bipartite Graph

In a complete bipartite graph Kᵣ,ₛ the vertices split into parts of sizes r and s. Every vertex in one part meets every vertex in the other. There are no edges within a part.

Count each cross-part pair once

m = rs

Choose one endpoint from the r vertices on the left and the other from the s on the right. This already counts each edge once. K₄,₄ has 16 edges, while K₈ on the same number of vertices has 28.

Equal and unequal part sizes

Count the neighbours of one vertex in each part. With both parts nonempty, Kᵣ,ₛ is regular precisely when the two parts have the same size.

12Triangle-Free Graph

A triangle-free graph contains no cycle of length three. In C₈, the two neighbours of any vertex are never adjacent to each other. That local observation certifies triangle-freeness at every vertex.

Test a candidate closing edge

Triangles added = |N(u) ∩ N(v)|

For a missing edge uv, each common neighbour w gives exactly one new triangle u–w–v–u. In C₈, vertices 0 and 2 share only vertex 1. Adding 0–2 therefore creates one triangle.

Triangle-free cycles with different colourings

A bipartition would force the two parts to alternate on every cycle, so every bipartite graph is triangle-free. The letter inside each vertex below identifies its colour class.

14Bridge

A bridge is an edge whose deletion increases the component count. Edge 3–4 is the only connection between two four-cycles. The same deletion test on a rim edge gives a different outcome.

A cycle supplies a detour

e is a bridge ⇔ e lies on no cycle

If e lies on a cycle, the rest of that cycle connects its endpoints after deletion. Conversely, any surviving path between its endpoints combines with e to form a cycle.

Test every edge of the original graph

Delete a candidate edge and compare c with the intact graph. An edge that was the only path between its endpoints is a bridge.

16Matching

A matching is a set of edges with no shared endpoint. It is perfect if every vertex is an endpoint of a selected edge. Here the gold edges cover all eight vertices exactly once.

Maximal, maximum and perfect

|M| ≤ ⌊n / 2⌋

A maximal matching cannot be extended by a single edge. A maximum matching has largest possible cardinality. Every perfect matching reaches the counting bound and is maximum. On the four-vertex path below, the middle edge alone is already maximal.

Can a maximal matching be improved?

17Regular Graph

A k-regular graph has degree k at every vertex. Write δ(G) and Δ(G) for the minimum and maximum vertex degrees. Regularity is exactly the condition δ(G) = Δ(G) = k.

Local degrees constrain the edge count

2m = nk

Summing n equal degrees gives nk. In particular, a cubic graph (k = 3) has an even number of vertices. Degree counts alone do not determine which pairs are adjacent.

Degrees two, three and four

Each row of the adjacency matrix of a k-regular graph sums to k. The three drawings show different ways to satisfy this local constraint.

Can degree sequences identify a graph?

18Wheel Graph

Form Wₙ by joining a new hub to all vertices on a cycle of length n − 1, where n ≥ 4 counts the hub. This drawing combines the rim of a cycle with the spokes of a star.

Count the rim and the spokes

m = 2(n − 1)

The rim contributes n − 1 edges and the hub contributes n − 1 more. Every rim vertex has degree three. At n = 4 the wheel is K₄, and the hub also has degree three.

Colouring even and odd rims

The hub sees every colour used on the rim. A new colour suffices for the hub, so χ(Wₙ) = χ(Cₙ₋₁) + 1. The vertex letters show a colouring attaining this bound.

19Planar Graph

A planar graph admits a plane drawing with edges meeting only at common endpoints. The original cube drawing has crossings. Rearranging the same eight vertices as two nested squares gives a crossing-free drawing.

Count regions after a plane embedding

n − m + f = 2

For a connected plane graph, f counts all faces, including the unbounded outer region. A plane tree has one face. Each edge added within a face to complete the graph splits that face in two, increasing m and f together.

20Non-Planar Graph

A nonplanar graph has no crossing-free drawing in the plane. The six non-isolated vertices in the original figure induce K₃,₃. If all eight vertices admitted a plane drawing, removing the two isolates would give a plane drawing of K₃,₃.

Test the obstruction on its own vertices

9 > 2 × 6 − 4

Suppose K₃,₃ had a plane drawing. Its face boundaries have length at least four because it is simple, bipartite and bridgeless. Counting the sides of each edge yields 4f ≤ 2m. Euler’s identity then gives m ≤ 2n − 4, which the nine edges violate.

Two edge-count obstructions

For a simple connected plane graph with n ≥ 3, counting face boundaries gives m ≤ 3n − 6. Bipartiteness sharpens the bound to 2n − 4. Violating the applicable bound proves nonplanarity; satisfying it alone cannot prove planarity.
